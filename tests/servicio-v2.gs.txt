/**
 * Crea las pestañas y las metas iniciales. Se puede ejecutar varias veces:
 * no borra ni cambia filas que ya existan.
 */
function setup() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.keys(TABLAS).forEach(function (nombre) {
    var h = ss.getSheetByName(nombre) || ss.insertSheet(nombre);
    asegurarEncabezados(h, nombre);
    h.setFrozenRows(1);
    // Solo las filas vacías: así Sheets no convierte «2026-10-09» en otra cosa.
    var ultima = Math.max(h.getLastRow(), 1);
    var max = h.getMaxRows();
    if (max > ultima) {
      encabezados(h).forEach(function (c, i) {
        if (esColumnaTexto(nombre, c)) h.getRange(ultima + 1, i + 1, max - ultima, 1).setNumberFormat('@');
      });
    }
  });
  var metas = ss.getSheetByName('metas');
  if (metas.getLastRow() < 2) metas.getRange(2, 1, METAS_INICIALES.length, 2).setValues(METAS_INICIALES);
  var vacia = ss.getSheetByName('Hoja 1') || ss.getSheetByName('Sheet1');
  if (vacia && vacia.getLastRow() === 0 && ss.getSheets().length > 1) ss.deleteSheet(vacia);
}

/** Diagnóstico: ejecútala desde el editor y mira el «Registro de ejecución». No cambia nada. */
function probar() {
  Logger.log('Versión del servicio: ' + VERSION + ' · zona de la hoja: ' + zona());
  Logger.log('Contraseña (TOKEN) configurada: ' + (leerToken().length >= 16 ? 'sí' : 'NO o muy corta (mínimo 16 caracteres)'));
  ['comidas', 'ejercicio', 'series_gym', 'medidas'].forEach(function (n) {
    var filas = leer(n);
    var fechas = filas.map(function (f) { return f.fecha; }).filter(Boolean).sort();
    Logger.log(n + ': ' + filas.length + ' filas' + (fechas.length ? ' · fechas de ' + fechas[0] + ' a ' + fechas[fechas.length - 1] : ''));
  });
}

function doGet() {
  return salida({ ok: true, servicio: 'mi-seguimiento', version: VERSION });
}

function doPost(e) {
  try {
    var pedido = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    if (!tokenValido(pedido.token)) return salida({ ok: false, error: 'No autorizado' });
    switch (pedido.accion) {
      case 'ping': return salida({ ok: true, version: VERSION, chat: !!claveApi() });
      case 'leer': return salida(leerTodo(pedido.desde));
      case 'dia': return salida({ ok: true, version: VERSION, datos: dia(pedido.fecha) });
      case 'agregar': return salida(conBloqueo(function () { return agregar(pedido.tabla, pedido.fila); }));
      case 'medidas': return salida(conBloqueo(function () { return guardarMedidas(pedido.fila); }));
      case 'borrar': return salida(conBloqueo(function () { return borrar(pedido.tabla, pedido.id); }));
      case 'chat': return salida(chat(pedido));
      default: return salida({ ok: false, error: 'Acción desconocida' });
    }
  } catch (err) {
    return salida({ ok: false, error: String(err && err.message ? err.message : err) });
  }
}

// ---------- utilidades ----------

function salida(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function leerToken() {
  return String(PropertiesService.getScriptProperties().getProperty('TOKEN') || '').trim();
}

function tokenValido(recibido) {
  var real = leerToken();
  if (typeof recibido !== 'string') return false;
  recibido = recibido.trim();
  if (real.length < 16 || recibido.length !== real.length) return false;
  var diff = 0;
  for (var i = 0; i < real.length; i++) diff |= real.charCodeAt(i) ^ recibido.charCodeAt(i);
  return diff === 0;
}

function conBloqueo(fn) {
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try { return fn(); } finally { lock.releaseLock(); }
}

function hoja(nombre) {
  var h = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(nombre);
  if (!h) throw new Error('Falta la pestaña "' + nombre + '". En Apps Script, ejecuta la función setup.');
  return h;
}

var zonaCache = null;
function zona() {
  if (!zonaCache) zonaCache = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone() || 'UTC';
  return zonaCache;
}

function dos(n) { n = String(n); return n.length < 2 ? '0' + n : n; }

function tipoColumna(nombre, col) {
  var t = (TABLAS[nombre] || []).filter(function (k) { return k.c === col; })[0];
  return t ? t.t : null;
}

function esColumnaTexto(nombre, col) {
  var t = tipoColumna(nombre, col);
  return t === 'fecha' || t === 'hora' || t === 'id';
}

/** Cualquier forma en que Sheets o una persona guarde una fecha → «AAAA-MM-DD». */
function normFecha(v) {
  if (v === '' || v === null || v === undefined) return null;
  if (v instanceof Date) return Utilities.formatDate(v, zona(), 'yyyy-MM-dd');
  if (typeof v === 'number') {
    // Número de serie de Sheets: días desde el 30/12/1899
    if (v > 20000 && v < 80000) return Utilities.formatDate(new Date(Date.UTC(1899, 11, 30) + Math.round(v) * 86400000), 'UTC', 'yyyy-MM-dd');
    return String(v);
  }
  var s = String(v).trim();
  var m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
  if (m) return m[1] + '-' + dos(m[2]) + '-' + dos(m[3]);
  m = s.match(/^(\d{1,2})[\/.\-](\d{1,2})[\/.\-](\d{4})$/); // día/mes/año
  if (m) return m[3] + '-' + dos(m[2]) + '-' + dos(m[1]);
  if (/^\d{4}-\d{2}-\d{2}T/.test(s)) {
    var d = new Date(s);
    if (!isNaN(d.getTime())) return Utilities.formatDate(d, zona(), 'yyyy-MM-dd');
  }
  return s;
}

/** Cualquier forma de hora → «HH:MM». */
function normHora(v) {
  if (v === '' || v === null || v === undefined) return null;
  if (v instanceof Date) return Utilities.formatDate(v, zona(), 'HH:mm');
  if (typeof v === 'number' && v >= 0 && v < 1) {
    var min = Math.round(v * 1440);
    return dos(Math.floor(min / 60) % 24) + ':' + dos(min % 60);
  }
  var m = String(v).trim().match(/^(\d{1,2}):(\d{2})/);
  return m ? dos(m[1]) + ':' + m[2] : String(v).trim();
}

function normalizar(v, tipo) {
  if (v === '' || v === null || v === undefined) return null;
  if (tipo === 'fecha') return normFecha(v);
  if (tipo === 'hora') return normHora(v);
  if (v instanceof Date) return Utilities.formatDate(v, zona(), "yyyy-MM-dd'T'HH:mm:ss");
  return v;
}

function encabezados(h) {
  var n = h.getLastColumn();
  if (n < 1) return [];
  return h.getRange(1, 1, 1, n).getValues()[0].map(function (x) { return String(x).trim(); });
}

/** Escribe los encabezados si la pestaña está vacía, o agrega al final los que falten. */
function asegurarEncabezados(h, nombre) {
  var cols = TABLAS[nombre].map(function (k) { return k.c; });
  var enc = encabezados(h);
  if (!enc.some(function (x) { return x; })) {
    h.getRange(1, 1, 1, cols.length).setValues([cols]).setFontWeight('bold');
    return cols;
  }
  cols.forEach(function (c) {
    if (enc.indexOf(c) < 0) {
      enc.push(c);
      h.getRange(1, enc.length).setValue(c).setFontWeight('bold');
    }
  });
  return enc;
}

/** Filas de una pestaña como objetos, con fechas y horas siempre como texto. */
function leer(nombre) {
  var valores = hoja(nombre).getDataRange().getValues();
  if (!valores.length) return [];
  var cols = valores[0].map(function (x) { return String(x).trim(); });
  return valores.slice(1)
    .filter(function (f) { return f.some(function (v) { return v !== '' && v !== null; }); })
    .map(function (f) {
      var o = {};
      cols.forEach(function (c, i) { if (c) o[c] = normalizar(f[i], tipoColumna(nombre, c)); });
      return o;
    });
}

/** Agrega una fila según los encabezados reales de la pestaña, con fecha, hora e id como texto. */
function escribirFila(h, nombre, obj) {
  var enc = asegurarEncabezados(h, nombre);
  var r = h.getLastRow() + 1;
  enc.forEach(function (c, i) {
    if (esColumnaTexto(nombre, c)) h.getRange(r, i + 1).setNumberFormat('@');
  });
  var valores = enc.map(function (c) { return obj[c] === null || obj[c] === undefined ? '' : obj[c]; });
  h.getRange(r, 1, 1, enc.length).setValues([valores]);
}

function ahora() { return new Date().toISOString(); }

/** Valida una fila contra el esquema y devuelve el objeto limpio. Acepta variantes comunes. */
function validar(nombre, fila) {
  if (!fila || typeof fila !== 'object') throw new Error('Datos vacíos');
  var out = {};
  TABLAS[nombre].forEach(function (k) {
    var v = fila[k.c];
    if (typeof v === 'string') v = v.trim();
    if (k.t === 'id') { out[k.c] = Utilities.getUuid(); return; }
    if (k.t === 'ts') { out[k.c] = ahora(); return; }
    if (v === undefined || v === null || v === '') {
      if (k.req) throw new Error('Falta ' + k.c);
      out[k.c] = k.t === 'bool' ? false : (k.def !== undefined ? k.def : null);
      return;
    }
    if (k.t === 'fecha') {
      v = normFecha(v);
      if (!/^\d{4}-\d{2}-\d{2}$/.test(v)) throw new Error('Fecha inválida (usa AAAA-MM-DD)');
    } else if (k.t === 'hora') {
      v = normHora(v);
      if (!/^\d{2}:\d{2}$/.test(v)) throw new Error('Hora inválida (usa HH:MM)');
    } else if (k.t === 'num') {
      v = Number(typeof v === 'string' ? v.replace(',', '.') : v);
      if (!isFinite(v)) throw new Error(k.c + ' debe ser un número');
      if (k.min !== undefined && v < k.min) throw new Error(k.c + ' es menor que ' + k.min);
      if (k.max !== undefined && v > k.max) throw new Error(k.c + ' es mayor que ' + k.max);
      if (k.ent && Math.floor(v) !== v) throw new Error(k.c + ' debe ser entero');
    } else if (k.t === 'bool') {
      v = v === true || /^(true|verdadero|s[ií]|1|yes)$/i.test(String(v));
    } else {
      v = String(v).slice(0, 500);
      if (k.op) {
        v = v.toLowerCase();
        if (k.op.indexOf(v) < 0) throw new Error(k.c + ' no es válido (usa: ' + k.op.join(', ') + ')');
      }
    }
    out[k.c] = v;
  });
  return out;
}

/** Todo lo que necesita la app: metas, medidas completas y el resto desde una fecha. */
function leerTodo(desde) {
  var d = normFecha(desde);
  var filtrar = function (filas) {
    return /^\d{4}-\d{2}-\d{2}$/.test(d || '') ? filas.filter(function (f) { return !f.fecha || f.fecha >= d; }) : filas;
  };
  var metas = {};
  leer('metas').forEach(function (m) { if (m.clave) metas[String(m.clave).trim()] = Number(m.valor); });
  return {
    ok: true,
    version: VERSION,
    chat: !!claveApi(),
    desde: d,
    metas: metas,
    medidas: leer('medidas'),
    comidas: filtrar(leer('comidas')),
    ejercicio: filtrar(leer('ejercicio')),
    series_gym: filtrar(leer('series_gym'))
  };
}

function agregar(nombre, datos) {
  if (!EDITABLES[nombre]) throw new Error('Tabla no permitida');
  var obj = validar(nombre, datos);
  escribirFila(hoja(nombre), nombre, obj);
  return { ok: true, id: obj.id, fecha: obj.fecha };
}

/** Una fila por día. Solo actualiza los campos que vienen llenos. */
function guardarMedidas(datos) {
  var obj = validar('medidas', datos);
  if (obj.peso_kg === null && obj.cintura_cm === null && obj.horas_sueno === null) throw new Error('Escribe al menos un dato');
  var h = hoja('medidas');
  var enc = asegurarEncabezados(h, 'medidas');
  var valores = h.getDataRange().getValues();
  var fi = enc.indexOf('fecha');
  for (var r = 1; r < valores.length; r++) {
    if (normFecha(valores[r][fi]) === obj.fecha) {
      ['peso_kg', 'cintura_cm', 'horas_sueno'].forEach(function (c) {
        if (obj[c] !== null) h.getRange(r + 1, enc.indexOf(c) + 1).setValue(obj[c]);
      });
      return { ok: true, actualizado: true, fecha: obj.fecha };
    }
  }
  escribirFila(h, 'medidas', obj);
  return { ok: true, actualizado: false, fecha: obj.fecha };
}

function borrar(nombre, id) {
  if (!EDITABLES[nombre]) throw new Error('Tabla no permitida');
  var h = hoja(nombre);
  var valores = h.getDataRange().getValues();
  var ci = valores[0].map(function (x) { return String(x).trim(); }).indexOf('id');
  for (var r = 1; r < valores.length; r++) {
    if (String(valores[r][ci]).trim() === String(id).trim()) { h.deleteRow(r + 1); return { ok: true }; }
  }
  throw new Error('No se encontró el registro');
}

/** Resumen de un día (lo usa el chat con API y las versiones anteriores de la app). */
function dia(fecha) {
  fecha = normFecha(fecha);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(fecha || '')) throw new Error('Fecha inválida');
  var t = leerTodo(null);
  var medidas = t.medidas.filter(function (m) { return m.fecha; }).sort(function (a, b) { return a.fecha < b.fecha ? -1 : 1; });
  var hastaHoy = medidas.filter(function (m) { return m.fecha <= fecha; });
  var conPeso = hastaHoy.filter(function (m) { return m.peso_kg !== null && m.peso_kg !== undefined; });
  var conCintura = hastaHoy.filter(function (m) { return m.cintura_cm !== null && m.cintura_cm !== undefined; });
  return {
    chat: t.chat,
    metas: t.metas,
    comidas: t.comidas.filter(function (c) { return c.fecha === fecha; })
      .sort(function (a, b) { return String(a.hora || '') < String(b.hora || '') ? -1 : 1; }),
    ejercicio: t.ejercicio.filter(function (e) { return e.fecha === fecha; }),
    series_gym: t.series_gym.filter(function (s) { return s.fecha === fecha; }),
    medida: medidas.filter(function (m) { return m.fecha === fecha; })[0] || null,
    ultimoPeso: conPeso.length ? conPeso[conPeso.length - 1] : null,
    ultimaCintura: conCintura.length ? conCintura[conCintura.length - 1] : null,
    pesos: conPeso.slice(-7).map(function (m) { return { fecha: m.fecha, peso_kg: m.peso_kg }; })
  };
}

function textoMetas() {
  var m = {};
  leer('metas').forEach(function (x) { m[x.clave] = x.valor; });
  var p = [];
  if (m.proteina_min_g != null && m.proteina_max_g != null) p.push('proteína de ' + m.proteina_min_g + ' a ' + m.proteina_max_g + ' g al día');
  if (m.kcal_base != null) {
    p.push('calorías base ' + m.kcal_base + ' kcal más el ejercicio del día' +
      (m.deficit_min_kcal != null && m.deficit_max_kcal != null ? ', con déficit de ' + m.deficit_min_kcal + ' a ' + m.deficit_max_kcal + ' kcal en días normales' : ''));

  return p.length ? 'Metas del usuario: ' + p.join('; ') + '.' : '';
}

    'No inventes datos que no dijo (no pongas peso si no lo dio). Si el mensaje no trae nada que registrar, contesta con una frase corta y no llames herramientas.',
    'Después de llamar las herramientas puedes añadir una frase corta; el sistema ya muestra lo guardado y los totales del día.'
  ].join('\n');
}

function aRegistro(a, pedido) {
  var copia = {};
  Object.keys(a).forEach(function (k) { copia[k] = a[k]; });
  var f = normFecha(a.fecha);
  copia.fecha = /^\d{4}-\d{2}-\d{2}$/.test(f || '') ? f : pedido.hoy;
  return copia;
}

function ejecutarHerramienta(nombre, a, pedido) {
  var r, f;
  if (nombre === 'registrar_comida') {
    f = aRegistro(a, pedido);
    if (!f.hora) f.hora = pedido.hora;
    r = conBloqueo(function () { return agregar('comidas', f); });
    return { tabla: 'comidas', id: r.id, fecha: f.fecha, texto: cap(f.tipo_comida) + ': ' + f.descripcion + ' · ' + redond(f.kcal) + ' kcal · ' + redond(f.proteina_g) + ' g proteína' + (f.estimado ? ' (estimado)' : '') };
  }
  if (nombre === 'registrar_medidas') {
    f = aRegistro(a, pedido);
    conBloqueo(function () { return guardarMedidas(f); });
    var partes = [];
    if (f.peso_kg != null) partes.push(f.peso_kg + ' kg');
    if (f.cintura_cm != null) partes.push('cintura ' + f.cintura_cm + ' cm');
    if (f.horas_sueno != null) partes.push(f.horas_sueno + ' h de sueño');
    return { tabla: 'medidas', id: null, fecha: f.fecha, texto: 'Medidas: ' + partes.join(' · ') };
  }
  if (nombre === 'registrar_ejercicio') {
    f = aRegistro(a, pedido);
    if (!f.fuente) f.fuente = 'manual';
    r = conBloqueo(function () { return agregar('ejercicio', f); });
    return { tabla: 'ejercicio', id: r.id, fecha: f.fecha, texto: cap(f.tipo) + (f.duracion_min ? ' · ' + f.duracion_min + ' min' : '') + (f.kcal != null ? ' · ' + redond(f.kcal) + ' kcal' : '') };
  }
  if (nombre === 'registrar_serie_gym') {
    f = aRegistro(a, pedido);
    r = conBloqueo(function () { return agregar('series_gym', f); });
    return { tabla: 'series_gym', id: r.id, fecha: f.fecha, texto: f.ejercicio + ' · serie ' + f.numero_serie + ': ' + f.peso_kg + ' kg × ' + f.repeticiones };
  }
  throw new Error('Herramienta desconocida');
}

function cap(s) { s = String(s || ''); return s.charAt(0).toUpperCase() + s.slice(1); }
function redond(n) { return n == null ? '?' : Math.round(Number(n) * 10) / 10; }

function llamarClaude(pedido, mensajes) {
  var cuerpo = {
    model: PropertiesService.getScriptProperties().getProperty('MODELO') || MODELO_POR_DEFECTO,
    max_tokens: 2048,
    system: promptSistema(pedido.hoy, pedido.hora, textoMetas()),
    messages: mensajes,
    tools: HERRAMIENTAS,
    output_config: { effort: 'low' },
    fallbacks: 'default'
  };
  var resp = UrlFetchApp.fetch(API_URL, {
    method: 'post',
    contentType: 'application/json',
    muteHttpExceptions: true,
    headers: { 'x-api-key': claveApi(), 'anthropic-version': '2023-06-01', 'anthropic-beta': 'server-side-fallback-2026-07-01' },
    payload: JSON.stringify(cuerpo)
  });
  var codigo = resp.getResponseCode();
  var json;
  try { json = JSON.parse(resp.getContentText()); } catch (e) { json = {}; }
  if (codigo !== 200) {
    var detalle = json && json.error && json.error.message ? ': ' + String(json.error.message).slice(0, 200) : '';
    throw new Error('El chat no pudo responder (código ' + codigo + ')' + detalle);
  }
  return json;
}

function chat(pedido) {
  if (!claveApi()) throw new Error('El chat no está activado: falta la propiedad ANTHROPIC_API_KEY.');
  var texto = String(pedido.texto || '').trim().slice(0, 1000);
  if (!texto) throw new Error('Escribe un mensaje.');
  if (!/^\d{4}-\d{2}-\d{2}$/.test(pedido.hoy || '')) throw new Error('Fecha inválida');
  if (!/^\d{2}:\d{2}$/.test(pedido.hora || '')) pedido.hora = '12:00';

  var mensajes = [];
  (Array.isArray(pedido.historial) ? pedido.historial.slice(-8) : []).forEach(function (m) {
    if (m && (m.rol === 'user' || m.rol === 'assistant') && typeof m.texto === 'string' && m.texto) {
      mensajes.push({ role: m.rol, content: m.texto.slice(0, 1000) });
    }
  });
  // la API exige que empiece con "user" y alterne
  while (mensajes.length && mensajes[0].role !== 'user') mensajes.shift();
  var limpio = [];
  mensajes.forEach(function (m) {
    if (limpio.length && limpio[limpio.length - 1].role === m.role) limpio[limpio.length - 1].content += '\n' + m.content;
    else limpio.push(m);
  });
  if (limpio.length && limpio[limpio.length - 1].role === 'user') limpio.pop();
  limpio.push({ role: 'user', content: texto });

  var r = llamarClaude(pedido, limpio);
  if (r.stop_reason === 'refusal') return { ok: true, respuesta: 'No pude procesar ese mensaje. Prueba escribiéndolo de otra forma.', guardados: [] };

  var textos = [], guardados = [], fallos = [], fechas = {};
  (r.content || []).forEach(function (b) {
    if (b.type === 'text' && b.text) textos.push(b.text.trim());
    if (b.type === 'tool_use') {
      try {
        var g = ejecutarHerramienta(b.name, b.input || {}, pedido);
        guardados.push({ tabla: g.tabla, id: g.id, texto: g.texto, fecha: g.fecha });
        if (g.tabla === 'comidas') fechas[g.fecha] = true;
      } catch (e) {
        fallos.push('No guardé ' + b.name.replace('registrar_', '').replace('_', ' ') + ': ' + e.message);
      }
    }
  });

  var partes = [];
  if (guardados.length) partes.push('Guardé:\n' + guardados.map(function (g) { return '• ' + g.texto; }).join('\n'));
  fallos.forEach(function (f) { partes.push(f); });
  Object.keys(fechas).forEach(function (f) { partes.push(totalesDelDia(f, f === pedido.hoy)); });
  if (textos.length) partes.push(textos.join('\n'));
  if (!partes.length) partes.push('No entendí qué registrar. ¿Puedes darme más detalle?');
  return { ok: true, respuesta: partes.join('\n\n'), guardados: guardados };
}

function totalesDelDia(fecha, esHoy) {
  var d = dia(fecha), kcal = 0, prot = 0;
  d.comidas.forEach(function (c) { kcal += Number(c.kcal) || 0; prot += Number(c.proteina_g) || 0; });
  var m = d.metas, pmin = m.proteina_min_g != null ? m.proteina_min_g : 130;
  var extra = 0;
  d.ejercicio.forEach(function (e) { extra += Number(e.kcal) || 0; });
  var base = m.kcal_base != null ? m.kcal_base : 2500;
  var def = ((m.deficit_min_kcal != null ? m.deficit_min_kcal : 200) + (m.deficit_max_kcal != null ? m.deficit_max_kcal : 300)) / 2;
  var meta = base + extra - def, quedan = Math.round(meta - kcal);
  return (esHoy ? 'Hoy' : 'Ese día') + ' llevas ' + Math.round(kcal) + ' kcal y ' + Math.round(prot) + ' g de proteína. ' +
    (quedan >= 0 ? 'Te quedan ' + quedan + ' kcal' : 'Te pasaste ' + (-quedan) + ' kcal') +
    (prot >= pmin ? ' y ya llegaste a tu meta de proteína.' : ' y te faltan ' + Math.round(pmin - prot) + ' g de proteína para la meta de ' + pmin + ' g.');
}
